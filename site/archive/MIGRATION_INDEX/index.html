# Check schema compliance
python3 << 'PYTHON'
import json
from pathlib import Path

required_keys = ['metadata', 'electrical_ratings', 'thermal_properties', 'switch', 'diode']
errors = []

for f in Path('transistors_merged').glob('*.json'):
    with open(f) as fp:
        data = json.load(fp)
    missing = [k for k in required_keys if k not in data]
    if missing:
        errors.append(f"{f.name}: missing {missing}")

print(f"Validated: {245 - len(errors)}/245 files OK")
if errors:
    for e in errors[:5]:
        print(f"  ❌ {e}")
PYTHON
