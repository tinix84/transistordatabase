    """Test core domain models."""
import pytest
from transistordatabase.core.models import (
    Transistor,
    TransistorMetadata,
    ElectricalRatings,
    ThermalProperties,
    ChannelCharacteristics,
    SwitchingLossData,
    GateChargeCurve,
    SOA,
    VoltageDependentCapacitance,
)


def test_transistor_creation():
    """Test basic transistor object creation."""
    metadata = TransistorMetadata(
        name="TEST_MOSFET",
        type="MOSFET",
        author="Test",
        manufacturer="TestCorp",
        housing_type="TO-247"
    )
    electrical = ElectricalRatings(
        v_abs_max=600.0,
        i_abs_max=100.0,
        i_cont=50.0,
        t_j_max=175.0
    )
    thermal = ThermalProperties(
        housing_area=1.0,
        cooling_area=0.5
    )

    transistor = Transistor(metadata, electrical, thermal)

    assert transistor.metadata.name == "TEST_MOSFET"
    assert transistor.electrical_ratings.v_abs_max == 600.0
    assert transistor.switch is not None
    assert transistor.diode is not None


def test_channel_characteristics_validation():
    """Test channel curve data validation."""
    # Valid data
    channel = ChannelCharacteristics(
        t_j=25.0,
        v_g=15.0,
        dataset_type="graph_i_v",
        graph_v_i=[[0, 1, 2], [0, 10, 20]]
    )
    assert channel.t_j == 25.0
    assert len(channel.graph_v_i[0]) == 3

    # Invalid: mismatched array lengths
    with pytest.raises(ValueError):
        ChannelCharacteristics(
            t_j=25.0,
            v_g=15.0,
            dataset_type="graph_i_v",
            graph_v_i=[[0, 1, 2], [0, 10]]  # Mismatch!
        )


def test_switching_loss_data():
    """Test switching loss data structure."""
    loss = SwitchingLossData(

import pytest
import tempfile
import shutil
from pathlib import Path
from transistordatabase.core.repository import JsonTransistorRepository

def temp_repo_dir():
    """Create temporary repository directory."""
    temp_dir = tempfile.mkdtemp()
    yield Path(temp_dir)
    shutil.rmtree(temp_dir)


@pytest.fixture
def sample_transistor():
    """Create sample transistor for testing."""
    metadata = TransistorMetadata(
        name="TEST_MOSFET_001",
        type="MOSFET",
        author="Test",

        housing_type="TO-247"
    )
    electrical = ElectricalRatings(
        v_abs_max=600.0,
        i_abs_max=100.0,
        i_cont=50.0,
        t_j_max=175.0
    )
    thermal = ThermalProperties(housing_area=1.0, cooling_area=0.5)
    return Transistor(metadata, electrical, thermal)


def test_repository_save_and_load(temp_repo_dir, sample_transistor):
    """Test saving and loading transistor."""
    repo = JsonTransistorRepository(str(temp_repo_dir))

    # Save
    repo.save(sample_transistor)

    # Load
    loaded = repo.load("TEST_MOSFET_001")

    assert loaded is not None
    assert loaded.metadata.name == "TEST_MOSFET_001"
    assert loaded.electrical_ratings.v_abs_max == 600.0


def test_repository_list_all(temp_repo_dir, sample_transistor):
    """Test listing all transistors."""
    repo = JsonTransistorRepository(str(temp_repo_dir))

    # Save multiple
    repo.save(sample_transistor)

    sample_transistor.metadata.name = "TEST_MOSFET_002"
    repo.save(sample_transistor)

    # List all
    all_transistors = repo.list_all()

    assert len(all_transistors) >= 2
    names = [t.metadata.name for t in all_transistors]
    assert "TEST_MOSFET_001" in names
    assert "TEST_MOSFET_002" in names


def test_repository_delete(temp_repo_dir, sample_transistor):
    """Test deleting transistor."""
    repo = JsonTransistorRepository(str(temp_repo_dir))

    # Save
    repo.save(sample_transistor)
    assert repo.load("TEST_MOSFET_001") is not None

    # Delete
    repo.delete("TEST_MOSFET_001")

    # Verify deleted
    with pytest.raises(FileNotFoundError):
        repo.load("TEST_MOSFET_001")


def test_repository_update_curve_data(temp_repo_dir, sample_transistor):
    """Test updating curve data."""
    repo = JsonTransistorRepository(str(temp_repo_dir))

    # Save initial
    repo.save(sample_transistor)

    # Load, modify, save
    transistor = repo.load("TEST_MOSFET_001")
    from transistordatabase.core.models import ChannelCharacteristics

    new_curve = ChannelCharacteristics(
        t_j=25.0,
        v_g=15.0,
        dataset_type="graph_i_v",
        graph_v_i=[[0, 1, 2], [0, 10, 20]]
    )
    transistor.switch.channel_data.append(new_curve)
    repo.save(transistor)

    # Reload and verify
    reloaded = repo.load("TEST_MOSFET_001")
    assert len(reloaded.switch.channel_data) == 1
    assert reloaded.switch.channel_data[0].t_j == 25.0

def test_load_legacy_json_format():
    """Test loading old JSON format."""
    # Create legacy format JSON
    legacy_data = {
        "name": "LEGACY_MOSFET",
        "type": "MOSFET",
        "author": "Legacy",
        "manufacturer": "LegacyCorp",
        "housing_type": "TO-247",
        "v_abs_max": 600.0,
        "i_abs_max": 100.0,
        "i_cont": 50.0,
        "t_j_max": 175.0
    }

    temp_dir = tempfile.mkdtemp()
    legacy_file = Path(temp_dir) / "LEGACY_MOSFET.json"

    with open(legacy_file, 'w') as f:
        json.dump(legacy_data, f)

    try:
        # Load using repository
        loader = JsonTransistorLoader(temp_dir)
        transistor = loader.load_from_json("LEGACY_MOSFET")

        assert transistor.metadata.name == "LEGACY_MOSFET"
        assert transistor.electrical_ratings.v_abs_max == 600.0

    finally:
        shutil.rmtree(temp_dir)
