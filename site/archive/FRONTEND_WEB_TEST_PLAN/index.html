    Frontend Web GUI Test Plan¶

npm install

    File: src/components/__tests__/CurveEditor.spec.js 

    import { describe, it, expect, beforeEach, vi } from 'vitest'
import { mount } from '@vue/test-utils'
import CurveEditor from '../CurveEditor.vue'

describe('CurveEditor.vue', () => {
  let wrapper

  beforeEach(() => {
    wrapper = mount(CurveEditor, {
      props: {
        curveType: 'channel',
        component: 'switch'
      }
    })
  })

  it('renders form fields', () => {
    expect(wrapper.find('input[type="number"]').exists()).toBe(true)
    expect(wrapper.find('textarea').exists()).toBe(true)
  })

  it('validates required fields', async () => {
    const saveButton = wrapper.find('button.btn-save')
    await saveButton.trigger('click')

    expect(wrapper.vm.errors.length).toBeGreaterThan(0)
  })

  it('parses CSV data correctly', async () => {
    const csvData = `0,0
1,10
2,20
3,30`

    await wrapper.vm.parseCsvData(csvData, 'voltage')
    expect(wrapper.vm.curveData.v_data).toEqual([0, 1, 2, 3])
  })

  it('validates voltage/current array lengths match', async () => {
    wrapper.vm.curveData.v_data = [0, 1, 2]
    wrapper.vm.curveData.i_data = [0, 10]  // Mismatch!

    await wrapper.vm.validateCurve()
    expect(wrapper.vm.errors).toContain('Voltage and current arrays must have same length')
  })

  it('handles file upload', async () => {
    const file = new File(['0,0\n1,10\n2,20'], 'curve.csv', { type: 'text/csv' })
    const input = wrapper.find('input[type="file"]')

    await input.setValue([file])
    await wrapper.vm.$nextTick()

    expect(wrapper.vm.uploadedFileName).toBe('curve.csv')
  })

  it('emits save event with correct data', async () => {
    wrapper.vm.curveData = {
      t_j: 25,
      v_g: 15,
      v_data: [0, 1, 2],
      i_data: [0, 10, 20]
    }

    await wrapper.vm.saveCurve()
    expect(wrapper.emitted('save')).toBeTruthy()
    expect(wrapper.emitted('save')[0][0]).toEqual(wrapper.vm.curveData)
  })

  it('clears form on cancel', async () => {
    wrapper.vm.curveData.t_j = 25
    wrapper.vm.curveData.v_data = [0, 1, 2]

    await wrapper.vm.cancel()
    expect(wrapper.vm.curveData.t_j).toBe(null)
    expect(wrapper.vm.curveData.v_data).toEqual([])
  })
})

    Test 2.2.3: API Client (api.js)¶

    File: src/services/__tests__/api.spec.js 

    import { describe, it, expect, beforeEach, vi } from 'vitest'
import axios from 'axios'
import { transistorApi } from '../api'

vi.mock('axios')

describe('TransistorAPI', () => {
  beforeEach(() => {
    vi.clearAllMocks()
  })

  it('getAll() fetches all transistors', async () => {
    const mockData = [{ metadata: { name: 'T1' } }]
    axios.get.mockResolvedValue({ data: mockData })

    const result = await transistorApi.getAll()
    expect(axios.get).toHaveBeenCalledWith('/api/transistors')
    expect(result).toEqual(mockData)
  })

  it('getById() fetches single transistor', async () => {
    const mockData = { metadata: { name: 'T1' } }
    axios.get.mockResolvedValue({ data: mockData })

    const result = await transistorApi.getById('T1')
    expect(axios.get).toHaveBeenCalledWith('/api/transistors/T1')
    expect(result).toEqual(mockData)
  })

  it('create() posts new transistor', async () => {
    const transistorData = { metadata: { name: 'T2' } }
    axios.post.mockResolvedValue({ data: { id: 'T2' } })

    const result = await transistorApi.create(transistorData)
    expect(axios.post).toHaveBeenCalledWith('/api/transistors', transistorData)
    expect(result).toEqual({ id: 'T2' })
  })

  it('addChannelCurve() posts curve data', async () => {
    const curveData = { t_j: 25, v_g: 15, v_data: [0, 1], i_data: [0, 10] }
    axios.post.mockResolvedValue({ data: { success: true } })

    const result = await transistorApi.addChannelCurve('T1', curveData, 'switch')
    expect(axios.post).toHaveBeenCalledWith(
      '/api/transistors/T1/curves/channel',
      expect.objectContaining({ ...curveData, component: 'switch' })
    )
    expect(result.success).toBe(true)
  })

  it('compareAdvanced() posts comparison request', async () => {
    const mockData = { channel_characteristics: { labels: [], datasets: [] } }
    axios.post.mockResolvedValue({ data: mockData })

    const result = await transistorApi.compareAdvanced(['T1', 'T2'], {})
    expect(axios.post).toHaveBeenCalledWith('/api/comparison/advanced', {
      transistor_ids: ['T1', 'T2'],
      config: {}
    })
    expect(result).toEqual(mockData)
  })

  it('importPLECS() uploads file', async () => {
    const file = new File(['<xml></xml>'], 'test.xml', { type: 'text/xml' })
    const mockResponse = { success: true, transistors_imported: 2 }
    axios.post.mockResolvedValue({ data: mockResponse })

    const result = await transistorApi.importPLECS(file)

    expect(axios.post).toHaveBeenCalled()
    const callArgs = axios.post.mock.calls[0]
    expect(callArgs[0]).toBe('/api/import/plecs')
    expect(callArgs[1]).toBeInstanceOf(FormData)
    expect(result).toEqual(mockResponse)
  })

  it('calculateBiela() posts model parameters', async () => {
    const params = { c_oss: 150e-12, q_g: 50e-9, v_dc: 400, i_load: 20 }
    const mockResult = { results: { e_on: 0.0001, e_off: 0.00008 } }
    axios.post.mockResolvedValue({ data: mockResult })

    const result = await transistorApi.calculateBiela(params)
    expect(axios.post).toHaveBeenCalledWith('/api/analytical/biela', params)
    expect(result).toEqual(mockResult)
  })
})
