    import os
import json
from pathlib import Path
from collections import defaultdict

def analyze_directory(base_path):
    """Analyze directory structure and file formats."""
    stats = {
        'total_files': 0,
        'by_format': defaultdict(int),
        'by_type': defaultdict(int),
        'by_manufacturer': defaultdict(int),
        'files': []
    }

    for root, dirs, files in os.walk(base_path):
        for file in files:
            if file.endswith(('.json', '.xml', '.csv')):
                full_path = os.path.join(root, file)
                ext = file.split('.')[-1]

                stats['total_files'] += 1
                stats['by_format'][ext] += 1
                stats['files'].append(full_path)

                # Try to extract type and manufacturer
                if ext == 'json':
                    try:
                        with open(full_path, 'r') as f:
                            data = json.load(f)
                            if 'metadata' in data:
                                transistor_type = data['metadata'].get('type', 'unknown')
                                manufacturer = data['metadata'].get('manufacturer', 'unknown')
                            elif 'type' in data:
                                transistor_type = data.get('type', 'unknown')
                                manufacturer = data.get('manufacturer', 'unknown')
                            else:
                                transistor_type = 'unknown'
                                manufacturer = 'unknown'

                            stats['by_type'][transistor_type] += 1
                            stats['by_manufacturer'][manufacturer] += 1
                    except Exception as e:
                        print(f"Error reading {full_path}: {e}")

    return stats

    import json
from pathlib import Path

def analyze_json_structure(file_path):
    """Analyze structure of JSON file."""
    with open(file_path, 'r') as f:
        data = json.load(f)

    def get_structure(obj, prefix=''):
        """Recursively get structure."""
        if isinstance(obj, dict):
            for key, value in obj.items():
                if isinstance(value, (dict, list)):
                    get_structure(value, f"{prefix}.{key}" if prefix else key)
                else:
                    print(f"{prefix}.{key} = {type(value).__name__}")
        elif isinstance(obj, list) and len(obj) > 0:
            print(f"{prefix}[] ({len(obj)} items)")
            get_structure(obj[0], f"{prefix}[0]")

    get_structure(data)

    4.1: Create Migration Framework¶

    # migrate_to_tdb.py

import json
import logging
from pathlib import Path
from typing import Dict, List, Any
from dataclasses import dataclass
from datetime import datetime

@dataclass
class MigrationResult:
    """Result of migrating one transistor."""
    source_file: str
    source_format: str
    success: bool
    transistor_id: str
    error: str = None
    warnings: List[str] = None
    data_quality_score: float = 0.0

class TransistorMigrator:
    """Main migration orchestrator."""

    def __init__(self):
        self.logger = logging.getLogger(__name__)
        self.results: List[MigrationResult] = []
        self.duplicates: Dict[str, List[str]] = {}
        self.conflicts: List[Dict] = []

    def migrate_all(self, sources: List[Path], output_dir: Path):
        """Migrate all transistors from all sources."""
        for source_path in sources:
            self.logger.info(f"Processing {source_path}")
            self.migrate_directory(source_path, output_dir)

        self.generate_report()

    def migrate_directory(self, source_dir: Path, output_dir: Path):
        """Migrate all files in a directory."""
        for file_path in source_dir.rglob('*'):
            if file_path.suffix in ['.json', '.xml', '.csv']:
                result = self.migrate_file(file_path, output_dir)
                self.results.append(result)

    def migrate_file(self, file_path: Path, output_dir: Path) -> MigrationResult:
        """Migrate a single file."""
        try:
            # Detect format
            if file_path.suffix == '.json':
                data = self.load_json(file_path)
                converted = self.convert_json_to_tdb(data)
            elif file_path.suffix == '.xml':
                data = self.load_xml(file_path)
                converted = self.convert_xml_to_tdb(data)
            elif file_path.suffix == '.csv':
                data = self.load_csv(file_path)
                converted = self.convert_csv_to_tdb(data)

            # Validate
            validation_result = self.validate_tdb_data(converted)

            # Check for duplicates
            transistor_id = converted['metadata']['name']
            if self.is_duplicate(transistor_id):
                self.handle_duplicate(transistor_id, converted, file_path)
            else:
                # Save to output
                output_path = output_dir / f"{transistor_id}.json"
                self.save_tdb_json(converted, output_path)

            return MigrationResult(
                source_file=str(file_path),
                source_format=file_path.suffix,
                success=True,
                transistor_id=transistor_id,
                warnings=validation_result['warnings'],
                data_quality_score=validation_result['score']
            )

        except Exception as e:
            self.logger.error(f"Failed to migrate {file_path}: {e}")
            return MigrationResult(
                source_file=str(file_path),
                source_format=file_path.suffix,
                success=False,
                transistor_id=None,
                error=str(e)
            )

    def convert_json_to_tdb(self, data: Dict) -> Dict:
        """Convert source JSON to TDB format."""
        # Check if already in TDB format
        if self.is_tdb_format(data):
            return data

        # Apply field mapping
        from field_mapping import apply_mapping
        converted = apply_mapping(data, 'source_to_tdb')

        # Add metadata
        converted.setdefault('metadata', {})
        converted['metadata']['import_source'] = 'migration'
        converted['metadata']['import_date'] = datetime.now().isoformat()

        return converted

    def is_duplicate(self, transistor_id: str) -> bool:
        """Check if transistor already exists."""
        output_path = Path(f"transistors/{transistor_id}.json")
        return output_path.exists()

    def handle_duplicate(self, transistor_id: str, new_data: Dict, source_file: Path):
        """Handle duplicate transistor - compare and flag for review."""
        existing_path = Path(f"transistors/{transistor_id}.json")
        with open(existing_path, 'r') as f:
            existing_data = json.load(f)

        # Compare completeness
        new_score = self.calculate_completeness(new_data)
        existing_score = self.calculate_completeness(existing_data)

        self.conflicts.append({
            'transistor_id': transistor_id,
            'existing_file': str(existing_path),
            'new_file': str(source_file),
            'existing_score': existing_score,
            'new_score': new_score,
            'recommendation': 'keep_new' if new_score > existing_score else 'keep_existing'
        })

    def calculate_completeness(self, data: Dict) -> float:
        """Calculate data completeness score (0-100)."""
        required_fields = [
            'metadata.name',
            'metadata.type',
            'metadata.manufacturer',
            'electrical_ratings.v_abs_max',
            'electrical_ratings.i_abs_max',
            'switch.channel_data',
        ]

        present = 0
        for field_path in required_fields:
            if self.has_field(data, field_path):
                present += 1

        return (present / len(required_fields)) * 100

    def validate_tdb_data(self, data: Dict) -> Dict:
        """Validate TDB data and generate warnings."""
        warnings = []

        # Check required fields
        if not data.get('metadata', {}).get('name'):
            warnings.append("Missing transistor name")

        # Check units and ranges
        v_abs_max = data.get('electrical_ratings', {}).get('v_abs_max', 0)
        if v_abs_max < 10 or v_abs_max > 10000:
            warnings.append(f"Suspicious v_abs_max: {v_abs_max}V")

        # Calculate score
        score = self.calculate_completeness(data)

        return {
            'valid': len(warnings) == 0,
            'warnings': warnings,
            'score': score
        }

    def generate_report(self):
        """Generate migration report."""
        report = {
            'timestamp': datetime.now().isoformat(),
            'total_processed': len(self.results),
            'successful': sum(1 for r in self.results if r.success),
            'failed': sum(1 for r in self.results if not r.success),
            'duplicates': len(self.conflicts),
            'average_quality_score': sum(r.data_quality_score for r in self.results if r.success) / max(1, sum(1 for r in self.results if r.success)),
            'results': [
                {
                    'file': r.source_file,

def convert_csv_to_tdb(csv_path: Path) -> Dict:
    """Convert CSV to TDB JSON."""
    import pandas as pd

    df = pd.read_csv(csv_path)

    # Map CSV columns to TDB format
    tdb_data = {
        'metadata': {
            'name': df['name'].iloc[0],
            'type': df['type'].iloc[0],
            'manufacturer': df.get('manufacturer', 'Unknown').iloc[0],
            'import_source': f'csv:{csv_path.name}',
            'import_date': datetime.now().isoformat()
        },
        'electrical_ratings': {
            'v_abs_max': float(df['v_ds'].iloc[0]) if 'v_ds' in df.columns else 0.0,
            'i_abs_max': float(df['i_d'].iloc[0]) if 'i_d' in df.columns else 0.0,
            'i_cont': float(df['i_d'].iloc[0]) if 'i_d' in df.columns else 0.0,
            't_j_max': 175.0  # Default
        },
        'thermal_properties': {
            'housing_area': 0.0,
            'cooling_area': 0.0
        },
        'switch': {
            'channel_data': [],
            'e_on_data': [],
            'e_off_data': [],
            'gate_charge_curves': [],
            'soa': []
        },
        'diode': {
            'channel_data': [],
            'e_rr_data': []
        }
    }

    return tdb_data

from transistordatabase.core.repository import JsonTransistorLoader
from transistordatabase.backend.concrete_services import ValidationService

def validate_all_merged():
    """Validate all merged transistor files."""
    merged_dir = Path('/home/tinix/claude_wsl/transistordatabase/transistors_merged')
    validator = ValidationService()

    results = []
    for json_file in merged_dir.glob('*.json'):
        try:
            # Load with TDB loader
            loader = JsonTransistorLoader(str(merged_dir))
            transistor = loader.load_from_json(json_file.stem)

            # Validate
            validation = validator.validate_transistor(transistor)

            results.append({
                'file': json_file.name,
                'valid': validation['valid'],
                'errors': validation['errors'],
                'warnings': validation['warnings']
            })

        except Exception as e:
            results.append({
                'file': json_file.name,
                'valid': False,
                'errors': [str(e)],
                'warnings': []
            })

    # Summary
    valid_count = sum(1 for r in results if r['valid'])
    total = len(results)

    print(f"\n✅ Validation complete: {valid_count}/{total} valid ({valid_count/total*100:.1f}%)")

    # Save detailed results
    with open('VALIDATION_REPORT.json', 'w') as f:
        json.dump(results, f, indent=2)

    return results

def generate_quality_report():
    """Generate comprehensive quality report."""
    with open('MIGRATION_REPORT.json', 'r') as f:
        migration = json.load(f)

    with open('VALIDATION_REPORT.json', 'r') as f:
        validation = json.load(f)

    report = {
        'summary': {
            'total_transistors': len(validation),
            'valid': sum(1 for v in validation if v['valid']),
            'invalid': sum(1 for v in validation if not v['valid']),
            'avg_quality_score': migration['average_quality_score']
        },
        'by_manufacturer': {},
        'by_type': {},
        'data_completeness': {
            'has_channel_curves': 0,
            'has_switching_losses': 0,
            'has_gate_charge': 0,
            'has_soa': 0,
            'has_thermal_model': 0
        },
        'issues': {
            'missing_fields': [],
            'suspicious_values': [],
            'incomplete_records': []
        }
    }

    # Analyze each transistor
    merged_dir = Path('/home/tinix/claude_wsl/transistordatabase/transistors_merged')
    for json_file in merged_dir.glob('*.json'):
        with open(json_file, 'r') as f:
            data = json.load(f)

        # Count by manufacturer
        mfr = data['metadata'].get('manufacturer', 'Unknown')
        report['by_manufacturer'][mfr] = report['by_manufacturer'].get(mfr, 0) + 1

        # Count by type
        transistor_type = data['metadata'].get('type', 'Unknown')
        report['by_type'][transistor_type] = report['by_type'].get(transistor_type, 0) + 1

        # Check completeness
        if data['switch'].get('channel_data'):
            report['data_completeness']['has_channel_curves'] += 1
        if data['switch'].get('e_on_data') or data['switch'].get('e_off_data'):
            report['data_completeness']['has_switching_losses'] += 1
        if data['switch'].get('gate_charge_curves'):
            report['data_completeness']['has_gate_charge'] += 1
        if data['switch'].get('soa'):
            report['data_completeness']['has_soa'] += 1

    with open('QUALITY_REPORT.json', 'w') as f:
        json.dump(report, f, indent=2)

    print("\n✅ Quality Report Generated")
    print(f"  Total Transistors: {report['summary']['total_transistors']}")
    print(f"  Valid: {report['summary']['valid']}")
    print(f"  Average Quality: {report['summary']['avg_quality_score']:.1f}%")
    print(f"\n  By Type: {report['by_type']}")
    print(f"  By Manufacturer: {report['by_manufacturer']}")

    return report
