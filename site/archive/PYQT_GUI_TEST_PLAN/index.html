    """Test for memory leaks in GUI."""
import pytest
import gc

from pytestqt.qtbot import QtBot
from transistordatabase.gui.gui import MainWindow


def test_repeated_window_creation_no_leak(qtbot):
    """Test that creating/destroying windows doesn't leak memory."""
    import psutil
    import os

    process = psutil.Process(os.getpid())
    initial_memory = process.memory_info().rss / 1024 / 1024  # MB

    # Create and destroy 10 windows
    for i in range(10):
        window = MainWindow()
        qtbot.addWidget(window)
        window.show()
        qtbot.wait(100)
        window.close()
        window.deleteLater()

        # Force garbage collection
        gc.collect()
        qtbot.wait(100)

    final_memory = process.memory_info().rss / 1024 / 1024  # MB
    memory_increase = final_memory - initial_memory

    # Memory should not increase by more than 50 MB
    assert memory_increase < 50, f"Memory increased by {memory_increase} MB (expected < 50 MB)"
